// Правила для читання/оновлення сповіщень із мобільного застосунку.
// Перевір, чи вже є `allow read` для НЕ-адмінів у твоїх поточних правилах —
// найімовірніша причина "вічного завантаження" сторінки сповіщень це саме
// відсутність дозволу на read/list для звичайного employee/team_lead.

match /notifications/{notificationId} {
  // Будь-який залогінений герой може читати колекцію (сам запит на
  // клієнті вже фільтрує лише релевантні йому документи через Filter.or).
  allow read: if request.auth != null;

  allow create, delete: if request.auth.token.role == 'admin';

  allow update: if request.auth.token.role == 'admin'
    || (request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['readBy']));
}

// Дозволяє герою писати ЛИШЕ своє поле fcmTokens у власному документі.
match /users/{userId} {
  // ...існуючі правила...

  allow update: if request.auth != null
    && request.auth.uid == userId
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['fcmTokens']);
}
